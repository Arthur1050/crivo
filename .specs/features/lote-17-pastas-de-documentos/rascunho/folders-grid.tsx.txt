"use client";

import { EllipsisVerticalIcon, FolderIcon } from "lucide-react";
import { Card } from "@astryxdesign/core/Card";
import { ClickableCard } from "@astryxdesign/core/ClickableCard";
import { DropdownMenu } from "@astryxdesign/core/DropdownMenu";
import { Grid } from "@astryxdesign/core/Grid";
import { Icon } from "@astryxdesign/core/Icon";
import { HStack, StackItem, VStack } from "@astryxdesign/core/Stack";
import { Heading, Text } from "@astryxdesign/core/Text";
import type { DocumentFolder } from "@/src/components/documents/mock-folders";
import { formatFileSize } from "@/src/lib/format";

interface FoldersGridProps {
  folders: DocumentFolder[];
}

function formatFileCount(count: number) {
  return count === 1 ? "1 arquivo" : `${count} arquivos`;
}

/**
 * Seção "Pastas" do gerenciador de documentos: cards clicáveis em grade
 * responsiva (até 4 colunas), cada um com nome, contagem, última atualização
 * e tamanho. Por ora alimentada por `MOCK_DOCUMENT_FOLDERS` — abrir e
 * renomear ganham comportamento quando o sistema de pastas existir.
 */
export function FoldersGrid({ folders }: FoldersGridProps) {
  return (
    <VStack gap={3}>
      <HStack gap={3} vAlign="center">
        <StackItem size="fill">
          <Heading level={2}>Pastas</Heading>
        </StackItem>
        <Text type="supporting">
          {folders.length === 1 ? "1 pasta" : `${folders.length} pastas`}
        </Text>
      </HStack>

      <Grid columns={{ minWidth: 220, max: 4 }} gap={3}>
        {folders.map((folder) => (
          <ClickableCard key={folder.id} label={`Pasta ${folder.name}`} padding={3}>
            <VStack gap={4}>
              <HStack gap={3} vAlign="center">
                <Card variant="muted" padding={2}>
                  <FolderIcon size={18} className="text-secondary" />
                </Card>
                <StackItem size="fill">
                  <VStack gap={0.5}>
                    <Text type="body" weight="semibold" maxLines={1}>
                      {folder.name}
                    </Text>
                    <Text type="supporting">{formatFileCount(folder.fileCount)}</Text>
                  </VStack>
                </StackItem>
                <DropdownMenu
                  button={{
                    label: `Ações da pasta ${folder.name}`,
                    icon: <EllipsisVerticalIcon size={16} />,
                    variant: "ghost",
                    size: "sm",
                    isIconOnly: true,
                  }}
                  hasChevron={false}
                  menuWidth={180}
                  items={[
                    { label: "Abrir pasta" },
                    { label: "Copiar link" },
                    { label: "Renomear" },
                  ]}
                />
              </HStack>
              <HStack gap={1.5} vAlign="center">
                <Icon icon="clock" size="xsm" color="secondary" />
                <StackItem size="fill">
                  <Text type="supporting" maxLines={1}>
                    {folder.updatedLabel}
                  </Text>
                </StackItem>
                <Text type="supporting">{formatFileSize(folder.sizeBytes)}</Text>
              </HStack>
            </VStack>
          </ClickableCard>
        ))}
      </Grid>
    </VStack>
  );
}
